/**
 * Invite service for the Supabase starter.
 *
 * Same domain logic as the SQLite edition, backed by Supabase.
 * Single-use hashed tokens, 7-day expiry, atomic claim, revoke.
 */
 
import { audit } from '../audit';
import { isRole, mayGrant, requirePermission, RbacError, type Role } from '../rbac';
import type { SupabaseClient } from '@supabase/supabase-js';
import { BillingError, type BillingAdapter } from '../billing/adapter';
 
/**
 * Throwable error for invitation operations.
 *
 * `code` is a stable machine-readable string.
 *
 * @throws The thrown instance is this class; inspect `code` to branch:
 *   - `'invalid_token'` — the token is unknown or the underlying query failed.
 *   - `'expired'` — the invite has passed its expiry.
 *   - `'revoked'` — the invite was revoked.
 *   - `'already_accepted'` — the invite (or a concurrent claim) was already used.
 *   - `'already_member'` — the user is already a member or the membership insert failed.
 *   - `'bad_role'` — the invite creation failed at the database layer.
 */
export class InviteError extends Error {
  constructor(
    public code:
      'invalid_token' | 'expired' | 'revoked' | 'already_accepted' | 'already_member' | 'bad_role',
    message: string,
  ) {
    super(message);
  }
}
 
const DEFAULT_TTL_DAYS = 7;
 
/**
 * Default lifetime of an invite, in milliseconds (7 days).
 */
export const INVITE_TTL_MS = DEFAULT_TTL_DAYS * 24 * 60 * 60 * 1000;
 
/**
 * Hash a token using SHA-256 (via Web Crypto API in Supabase Edge Runtime).
 */
async function sha256(token: string): Promise<string> {
  const encoder = new TextEncoder();
  const data = encoder.encode(token);
  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  return hashArray.map((b) => b.toString(16).padStart(2, '0')).join('');
}
 
/**
 * Generate a cryptographically random, URL-safe invite token.
 *
 * 32 random bytes are base64-encoded with URL-safe substitutions and no
 * padding. Only the SHA-256 hash of the token is ever stored (see
 * {@link createInvite}); never log or persist the raw token.
 *
 * @returns A URL-safe token string.
 * @example
 * ```ts
 * const token = newInviteToken(); // e.g. "yhUQf4mKxV...-DSw2"
 * ```
 */
export function newInviteToken(): string {
  const array = new Uint8Array(32);
  crypto.getRandomValues(array);
  return btoa(String.fromCharCode(...array))
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=/g, '');
}
 
/**
 * Create a new invite for an org.
 *
 * The actor must hold `members.invite` and may only grant a role strictly
 * below their own rank. Only the SHA-256 hash of the token is persisted; the
 * raw token is returned to the caller exactly once. An `invite.created` audit
 * entry is written (without the token).
 *
 * @param supabase A Supabase client (service role, bypasses RLS — authorize in
 *   the caller before invoking).
 * @param input The invite details — `orgId`, actor, optional scoped `email`,
 *   the `role` to grant, and an optional `ttlDays` override (defaults to
 *   {@link INVITE_TTL_MS}).
 * @returns The created invite row id, the raw token (returned once, hash-only
 *   on disk), and the expiry as epoch milliseconds.
 * @throws {@link RbacError} with code `'bad_role'` when `role` is invalid, or
 *   `'hierarchy_violation'` when the actor cannot grant the role, or
 *   `'forbidden'` when the actor lacks `members.invite`.
 * @throws {@link InviteError} with code `'bad_role'` when the insert fails.
 */
export async function createInvite(
  supabase: SupabaseClient,
  input: {
    orgId: string;
    actorUserId: string;
    actorRole: Role;
    email?: string;
    role: string;
    ttlDays?: number;
  },
): Promise<{ id: string; token: string; expiresAtMs: number }> {
  requirePermission(input.actorRole, 'members.invite');
  const email = input.email?.trim().toLowerCase();
  if (!isRole(input.role)) {
    throw new RbacError('bad_role', `Unknown role '${input.role}'.`);
  }
  if (!mayGrant(input.actorRole, input.role)) {
    throw new RbacError(
      'hierarchy_violation',
      `A ${input.actorRole} cannot grant the '${input.role}' role.`,
    );
  }
 
  const ttlDays = input.ttlDays ?? DEFAULT_TTL_DAYS;
  const token = newInviteToken();
  const tokenHash = await sha256(token);
  const expiresAt = new Date(Date.now() + ttlDays * 24 * 60 * 60 * 1000).toISOString();
 
  const { data, error } = await supabase
    .from('invites')
    .insert({
      org_id: input.orgId,
      token_hash: tokenHash,
      role: input.role,
      email: email || null,
      invited_by: input.actorUserId,
      expires_at: expiresAt,
    })
    .select()
    .single();
 
  if (error) {
    throw new InviteError('bad_role', `Failed to create invite: ${error.message}`);
  }
 
  // NOTE: the raw token is never written to the audit log.
  await audit(supabase, {
    orgId: input.orgId,
    actorUserId: input.actorUserId,
    action: 'invite.created',
    targetType: 'invite',
    targetId: data.id,
    metadata: { role: input.role, email },
  });
 
  return { id: data.id, token, expiresAtMs: new Date(expiresAt).getTime() };
}
 
/**
 * List the pending invites for an org (not accepted, not revoked, not expired).
 *
 * @param supabase A Supabase client.
 * @param orgId The org whose invites to list.
 * @returns An array of pending invites with role, optional email, and epoch-millisecond
 *   timestamps.
 * @throws {@link InviteError} with code `'invalid_token'` when the query fails.
 */
export async function listPendingInvites(supabase: SupabaseClient, orgId: string) {
  const { data, error } = await supabase
    .from('invites')
    .select('id, role, email, expires_at, created_at')
    .eq('org_id', orgId)
    .is('accepted_at', null)
    .is('revoked_at', null)
    .gt('expires_at', new Date().toISOString())
    .order('created_at', { ascending: false });
 
  if (error) {
    throw new InviteError('invalid_token', `Failed to list invites: ${error.message}`);
  }
 
  return (data || []).map((row: any) => ({
    id: row.id,
    role: row.role,
    email: row.email,
    expiresAtMs: new Date(row.expires_at).getTime(),
    createdAtMs: new Date(row.created_at).getTime(),
  }));
}
 
/**
 * Revoke a pending invite.
 *
 * The actor must hold `invites.revoke`. A `invite.revoked` audit entry is
 * written. Revoking an invite that is already used, already revoked, or not
 * in the org fails the update gate and is an error.
 *
 * @param supabase A Supabase client.
 * @param input The operation context — `orgId`, acting user/role, and the
 *   `inviteId` to revoke.
 * @returns Resolves on success.
 * @throws {@link RbacError} with code `'forbidden'` when the actor lacks
 *   `invites.revoke`.
 * @throws {@link InviteError} with code `'invalid_token'` when the invite is
 *   not found, already used, or already revoked.
 */
export async function revokeInvite(
  supabase: SupabaseClient,
  input: { orgId: string; actorRole: Role; actorUserId: string; inviteId: string },
): Promise<void> {
  requirePermission(input.actorRole, 'invites.revoke');
 
  const { data, error } = await supabase
    .from('invites')
    .update({ revoked_at: new Date().toISOString() })
    .eq('id', input.inviteId)
    .eq('org_id', input.orgId)
    .is('revoked_at', null)
    .is('accepted_at', null)
    .select()
    .single();
 
  if (error || !data) {
    throw new InviteError('invalid_token', 'Invite not found, already used, or already revoked.');
  }
 
  await audit(supabase, {
    orgId: input.orgId,
    actorUserId: input.actorUserId,
    action: 'invite.revoked',
    targetType: 'invite',
    targetId: input.inviteId,
  });
}
 
/**
 * Read-only preview for the landing page of an invite link — never mutates.
 */
export async function peekInvite(
  supabase: SupabaseClient,
  token: string,
): Promise<
  | { state: 'invalid' }
  | { state: 'expired' }
  | { state: 'revoked' }
  | { state: 'accepted' }
  | { state: 'valid'; orgName: string; role: Role; expiresAtMs: number }
> {
  const tokenHash = await sha256(token);
 
  const { data: invite } = await supabase
    .from('invites')
    .select('*')
    .eq('token_hash', tokenHash)
    .limit(1)
    .single();
 
  if (!invite) return { state: 'invalid' };
  if (invite.revoked_at) return { state: 'revoked' };
  if (invite.accepted_at) return { state: 'accepted' };
  if (new Date(invite.expires_at) <= new Date()) return { state: 'expired' };
 
  const { data: org } = await supabase
    .from('organizations')
    .select('name')
    .eq('id', invite.org_id)
    .limit(1)
    .single();
 
  return {
    state: 'valid',
    orgName: org?.name ?? '',
    role: invite.role as Role,
    expiresAtMs: new Date(invite.expires_at).getTime(),
  };
}
 
/**
 * Atomic single-use claim: the UPDATE's WHERE clause is the concurrency gate,
 * so two people hitting the same link cannot both get in.
 */
export async function acceptInvite(
  supabase: SupabaseClient,
  input: { token: string; userId: string; billing: BillingAdapter },
): Promise<{ orgId: string; orgName: string; role: Role }> {
  const tokenHash = await sha256(input.token);
 
  const { data: invite } = await supabase
    .from('invites')
    .select('*')
    .eq('token_hash', tokenHash)
    .limit(1)
    .single();
 
  if (!invite) throw new InviteError('invalid_token', 'This invite link is not valid.');
  if (invite.revoked_at) throw new InviteError('revoked', 'This invite was revoked.');
  if (new Date(invite.expires_at) <= new Date())
    throw new InviteError('expired', 'This invite has expired.');
  if (invite.accepted_at)
    throw new InviteError('already_accepted', 'This invite has already been used.');
 
  // Check if user is already a member
  const { data: existing } = await supabase
    .from('memberships')
    .select('id')
    .eq('org_id', invite.org_id)
    .eq('user_id', input.userId)
    .limit(1)
    .single();
 
  if (existing)
    throw new InviteError('already_member', 'You are already a member of this organization.');
 
  // Check seat availability BEFORE the single-use claim, so a full org never
  // burns an invite it cannot honor. The billing adapter decides the limit.
  const { data: memberRows, error: countError } = await supabase
    .from('memberships')
    .select('id')
    .eq('org_id', invite.org_id);
 
  if (countError) {
    throw new BillingError('adapter_error', `Failed to count seats: ${countError.message}`);
  }
  await input.billing.assertSeatAvailable(invite.org_id, memberRows?.length ?? 0);
 
  // Atomic single-use claim
  const { data: claimed, error: claimError } = await supabase
    .from('invites')
    .update({
      accepted_at: new Date().toISOString(),
      accepted_by: input.userId,
    })
    .eq('id', invite.id)
    .is('accepted_at', null)
    .is('revoked_at', null)
    .select()
    .single();
 
  if (claimError || !claimed) {
    throw new InviteError('already_accepted', 'This invite has already been used.');
  }
 
  // Create membership
  const { error: memberError } = await supabase.from('memberships').insert({
    org_id: invite.org_id,
    user_id: input.userId,
    role: invite.role,
  });
 
  if (memberError) {
    throw new InviteError('already_member', `Failed to create membership: ${memberError.message}`);
  }
 
  // Get org name for response
  const { data: org } = await supabase
    .from('organizations')
    .select('name')
    .eq('id', invite.org_id)
    .limit(1)
    .single();
 
  await audit(supabase, {
    orgId: invite.org_id,
    actorUserId: input.userId,
    action: 'invite.accepted',
    targetType: 'invite',
    targetId: invite.id,
    metadata: { role: invite.role },
  });
 
  return { orgId: invite.org_id, orgName: org?.name ?? '', role: invite.role as Role };
}
